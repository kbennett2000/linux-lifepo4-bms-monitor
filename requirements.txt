# Core Bluetooth Low Energy library
bleak>=3.0.2

# BMS drivers (JBD/Jiabaida, ECO-WORTHY, and ~45 others).
#
# 0.28.0 is a floor, not a preference. Every release between 0.23 and 0.28 fixed
# something this project was hitting on a flaky adapter:
#   0.24  a hard timeout around connect, because a BlueZ D-Bus stall could hang the
#         poll loop indefinitely — bleak_retry_connector's own timeouts do not fire
#         for it; and disconnect(reset=True) now closes stale connections
#   0.27  an operation lock, so a read and a teardown cannot overlap
# It also replaced the `keep_alive=` constructor argument with a `BMSConfig` object;
# bms_driver detects both, so an older install keeps working.
aiobmsble>=0.28.0

# Lightweight web framework for the beautiful dashboard
flask>=3.0.0

# Recovers a wedged BLE adapter in-process (power-cycles the controller) when it
# stops responding after hours of scanning, instead of needing a manual
# `systemctl restart bluetooth` / reboot. Pulls in bluetooth-adapters, which we
# also use to discover the adapter's hci index + MAC.
bluetooth-auto-recovery>=1.6.0
